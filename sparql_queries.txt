# SELECT Q1 — Liste tous les étudiants avec filière et niveau
# Cas d'usage : tableau de bord administrateur
# ══════════════════════════════════════════════════════════════════════════════
PREFIX lod:    <https://data.lod-school.ma/ontology#>
PREFIX schema: <https://schema.org/>
PREFIX rdfs:   <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?etudiant ?niveau ?filiere ?ville ?mention
WHERE {
    ?etudiant a lod:Student ;
              lod:level ?niveau ;
              lod:enrolledIn ?prog ;
              schema:addressLocality ?ville ;
              lod:mention ?mention .
    ?prog rdfs:label ?filiere .
}
ORDER BY ?filiere ?niveau
LIMIT 50

---

# ══════════════════════════════════════════════════════════════════════════════
# SELECT Q2 — Compétences d'un étudiant donné
# Cas d'usage : page profil étudiant
# ══════════════════════════════════════════════════════════════════════════════
PREFIX lod:  <https://data.lod-school.ma/ontology#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>

SELECT ?etudiant ?competence ?categorie
WHERE {
    ?etudiant a lod:Student ;
              lod:hasSkill ?skill .
    ?skill skos:prefLabel ?competence ;
           skos:broader ?cat .
    ?cat skos:prefLabel ?categorie .
}
ORDER BY ?etudiant ?categorie
LIMIT 100

# ══════════════════════════════════════════════════════════════════════════════
# SELECT Q5 — Matching étudiant ↔ offres (compétences communes)
# Cas d'usage : recommandation de stages pour un étudiant
# ══════════════════════════════════════════════════════════════════════════════
PREFIX lod:  <https://data.lod-school.ma/ontology#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX schema: <https://schema.org/>

SELECT ?etudiant ?offre ?titre ?entreprise
       (COUNT(?skillCommun) AS ?nb_competences_communes)
WHERE {
    ?etudiant a lod:Student ;
              lod:hasSkill ?skillCommun .
    ?offre a lod:InternshipOffer ;
           schema:title ?titre ;
           lod:requiresSkill ?skillCommun ;
           lod:postedBy ?comp ;
           lod:status "Ouverte"@fr .
    ?comp schema:name ?entreprise .
}
GROUP BY ?etudiant ?offre ?titre ?entreprise
ORDER BY DESC(?nb_competences_communes)
LIMIT 20

---

# SELECT Q11 — Toutes les compétences par catégorie SKOS

# ══════════════════════════════════════════════════════════════════════════════

# ══════════════════════════════════════════════════════════════════════════════
# SELECT Q14 — Étudiants par niveau avec nombre de compétences
# Cas d'usage : statistiques par niveau
# ══════════════════════════════════════════════════════════════════════════════
PREFIX lod: <https://data.lod-school.ma/ontology#>

SELECT ?niveau (COUNT(DISTINCT ?etudiant) AS ?nb_etudiants)
              (AVG(?nb_skills) AS ?moy_competences)
WHERE {
    {
        SELECT ?etudiant ?niveau (COUNT(?skill) AS ?nb_skills)
        WHERE {
            ?etudiant a lod:Student ;
                      lod:level ?niveau ;
                      lod:hasSkill ?skill .
        }
        GROUP BY ?etudiant ?niveau
    }
}
GROUP BY ?niveau
ORDER BY ?niveau
